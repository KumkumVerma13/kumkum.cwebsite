Question 2: Enter an array from user and print the array in reverse order ?>

SOLUTION:
#include < stdio.h >
#include < conio.h >
void main() {
int numbers[5];
int i;
printf("Enter 5 numbers:\n");
for (i = 0; i < 5; i++) {
printf("Number : ", i + 1);
scanf("%d", &numbers[i]);
}
printf("\nThe array in reverse order is:\n");
for (i = 4; i >= 0; i--) {
printf("%d ", numbers[i]);
}
printf("\n");
}

OUTPUT:
Enter 5 numbers:
Number : 10
Number : 20
Number : 30
Number : 40
Number : 50

The array in reverse order is:
50 40 30 20 10
