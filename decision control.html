Question 13: Enter a number from the user if the user enters 30 then print the month in which we have 30 days. If the enter 31 then print the month 31 days for 28, 29 print the month else everything is invalid ?

SOLUTION:
#include < stdio.h >
#include < conio.h >
void main() {
  int days;
  printf("Enter number of days (28-31): ");
  scanf("%d", &days);
switch (days) { 
        case 28:
            printf("Month of 28 days: February\n");
            break;
        case 29: 
            printf("Month of 29 days: February (leap year)\n");
            break;
        case 30: 
            printf("Months of 30 days: April, June, September, November\n");
            break;
        case 31: 
            printf("Months of 31 days: January, March, May, July, August, October, December\n");
            break;
        default:
            printf("Invalid choice\n");
            break;
    }
}

OUTPUT:
Enter number of days (28-31): 31
Months of 31 days: January, March, May, July, August, October, December
