Question 1:Enter a file name from the user and the data from the user and put it in file.

Solution:
#include< stdio.h >
#include< conio.h >

void main()
{
    FILE *fp;
    char fname[30], data[100];

    clrscr();

    printf("Enter file name: ");
    scanf("%s", fname);

    fp = fopen(fname, "w");

Enter data: Hello World
Data saved successfully.

Question2:Enter the file name from the user and the data from the user and put it in file. But data should be in capital letters

Solution
#include < stdio.h >
#include < conio.h >

void main()
{
    FILE *fp;
    char fname[20], data[50];
    int i;

    clrscr();

    printf("Enter file name: ");
    scanf("%s", fname);

    fp = fopen(fname, "w");

    printf("Enter data: ");
    scanf("%s", data);

    for(i = 0; data[i] != '\0'; i++)
    {
        if(data[i] >= 'a' && data[i] <= 'z')
        {
            data[i] = data[i] - 32;
        }
    }

    fprintf(fp, "%s", data);

    fclose(fp);

    printf("Data saved successfully.");
}

Output
Enter file name: first.txt
Enter data: hello

Data saved successfully.

Question 3: Enter the file name from the user and the data from the user till the user wants to enter.

Solution
#include < stdio.h >
#include < conio.h >

void main()
{
    FILE *fp;
    char fname[20], data[50];
    int ch;

    clrscr();

    printf("Enter file name: ");
    scanf("%s", fname);

    fp = fopen(fname, "w");

    ch = 1;

    while(ch == 1)
    {
        printf("Enter data: ");
        scanf(" %[^\n]", data);

        fprintf(fp, "%s\n", data);

        printf("Enter 1 to continue or 0 to stop: ");
        scanf("%d", &ch);
    }

    fclose(fp);

    printf("Data saved successfully.");
}

Output
Enter file name: first.txt
Enter data: Hello World
Enter 1 to continue or 0 to stop: 1
Enter data: Welcome to C
Enter 1 to continue or 0 to stop: 0

Data saved successfully.

Question 4: Enter two files from the user and concatenate into 3rd file.

Solution
#include < stdio.h >
#include < conio.h >

void main()
{
    FILE *fp, *fp1, *fp2;
    char a[20], b[20], cfile[20], data[50];
    int c;

    clrscr();

    printf("Enter first file name: ");
    scanf("%s", a);

    fp = fopen(a, "w");
    printf("Enter first data: ");
    scanf(" %[^\n]", data);
    fprintf(fp, "%s\n", data);
    fclose(fp);

    printf("Enter second file name: ");
    scanf("%s", b);

    fp1 = fopen(b, "w");
    printf("Enter second data: ");
    scanf(" %[^\n]", data);
    fprintf(fp1, "%s\n", data);
    fclose(fp1);

    printf("Enter third file name: ");
    scanf("%s", cfile);

    fp = fopen(a, "r");
    fp1 = fopen(b, "r");
    fp2 = fopen(cfile, "w");

    c = fgetc(fp);

    while(c != -1)
    {
        fputc(c, fp2);
        c = fgetc(fp);
    }

    c = fgetc(fp1);

    while(c != -1)
    {
        fputc(c, fp2);
        c = fgetc(fp1);
    }

    fclose(fp);
    fclose(fp1);
    fclose(fp2);

    printf("Files joined successfully.");
}

Output
Enter first file name: a.txt
Enter first data: Hello World

Enter second file name: b.txt
Enter second data: Welcome to C

Enter third file name: c.txt

Files joined successfully.

Question 5: Enter the file name from the user and the data from the user and rename that file. But rename should be by user.

Solution
#include < stdio.h >
#include < conio.h >

void main()
{
    FILE *fp, *fp1;
    char fname[20], newname[20], data[50];
    int ch, c;

    clrscr();

    printf("Enter file name: ");
    scanf("%s", fname);

    fp = fopen(fname, "w");

    ch = 1;

    while(ch == 1)
    {
        printf("Enter data: ");
        scanf(" %[^\n]", data);

        fprintf(fp, "%s\n", data);

        printf("Enter 1 to continue or 0 to stop: ");
        scanf("%d", &ch);
    }

    fclose(fp);

    printf("Enter new file name: ");
    scanf("%s", newname);

    fp = fopen(fname, "r");
    fp1 = fopen(newname, "w");

    c = fgetc(fp);

    while(c != -1)
    {
        fputc(c, fp1);
        c = fgetc(fp);
    }

    fclose(fp);
    fclose(fp1);

    printf("File copied successfully.");
}

Output
Enter file name: first.txt
Enter data: Hello World
Enter 1 to continue or 0 to stop: 1
Enter data: Welcome to C
Enter 1 to continue or 0 to stop: 0

Enter new file name: second.txt

File copied successfully.

Question 6: create a file and count no. of char in a file.

Solution
#include < stdio.h >
#include < conio.h >

void main()
{
    FILE *fp;
    char name[20], data[50];
    int c, count = 0;

    clrscr();

    printf("Enter file name: ");
    scanf("%s", name);

    fp = fopen(name, "w");

    printf("Enter data: ");
    scanf(" %[^\n]", data);

    fprintf(fp, "%s", data);
    fclose(fp);

    fp = fopen(name, "r");

    c = fgetc(fp);

    while(c != -1)
    {
        count++;
        c = fgetc(fp);
    }

    fclose(fp);

    printf("Total characters = %d", count);
}

Output
Enter file name: abc.txt
Enter data: Hello World
Total characters = 11

Question 7: create a file and count no. of words in a file.

Solution
#include < stdio.h >
#include < conio.h >

void main()
{
    FILE *fp;
    char name[20], data[50];
    int c, count = 1;

    clrscr();

    printf("Enter file name: ");
    scanf("%s", name);

    fp = fopen(name, "w");

    printf("Enter data: ");
    scanf(" %[^\n]", data);

    fprintf(fp, "%s", data);
    fclose(fp);

    fp = fopen(name, "r");

    c = fgetc(fp);

    while(c != -1)
    {
        if(c == ' ')
            count++;

        c = fgetc(fp);
    }

    fclose(fp);

    printf("Total words = %d", count);
}

Output
Enter file name: abc.txt
Enter data: Hello World bye

Total words = 3
