Question 2: Write a program to enter 10 number from the user add the number and display the answer ?

SOLUTION:
#include < stdio.h >
int addNumbers() {
    int sum = 0;
    int num;
    
    printf("Enter 10 numbers:\n");
    
    for (int i = 1; i <= 10; i++) {
        printf("Number %d: ", i);
        scanf("%d", &num);
        sum = sum + num;
    }
    
    return sum; 
}

int main() {
    int totalSum = addNumbers(); 
    
    printf("\nThe total sum is: %d\n", totalSum);

Question 5: Swap numbers using Call by address ?
SOLUTION
#include < stdio.h >
void swap(int *a, int *b) {
    int value;
    value = *a;   
    *a = *b;     
    *b = value; 
}
int main() {
    int x = 10;
    int y = 20;

    printf("Before swapping: x = %d, y = %d\n", x, y);
    swap(&x, &y);

    printf("After swapping: x = %d, y = %d\n", x, y);
}

OUTPUT
Before swapping: x = 10, y = 20
After swapping: x = 20, y = 10
