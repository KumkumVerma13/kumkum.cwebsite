Question 1: Enter the four number from user and print it into matrix

SOLUTION
#include < stdio.h >
int main() {
    int num1, num2, num3, num4;
    printf("Enter 4 numbers: ");
    scanf("%d", &num1);
    scanf("%d", &num2);
    scanf("%d", &num3);
    scanf("%d", &num4);
    
     printf("\nMatrix:\n");
    printf("%d  %d\n", num1, num2);
    printf("%d  %d\n", num3, num4);
}

OUTPUT

Matrix:

30  40

Question 2: Enter the matrix from and transpose them ?

SOLUTION:
#include < stdio.h >
int main() {
    int matrix[2][2], i, j;

    printf("Enter 4 numbers for matrix:\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            scanf("%d", &matrix[i][j]);
        }
    }
    printf("\nOriginal Matrix:\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            printf("%d ", matrix[i][j]);
        }
        printf("\n");
    }
    printf("\nTransposed Matrix:\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            printf("%d ", matrix[j][i]); 
        }
        printf("\n");
    }
}

OUTPUT
Enter 4 numbers for matrix:
1 2
3 4

Original Matrix:
1 2 
3 4 

Transposed Matrix:
1 3 
2 4 

Question 3: Enter 2 matrix from user and add them ?

SOLUTION
#include < stdio.h >
int main() {
    int a[2][2], b[2][2], sum[2][2];
    int i, j;
    printf("Enter 4 numbers for the first matrix:\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            scanf("%d", &a[i][j]);
        }
    }
    printf("Enter 4 numbers for the second matrix:\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            scanf("%d", &b[i][j]);
        }
    }
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            sum[i][j] = a[i][j] + b[i][j];
        }
    }
    printf("\nThe Sum is:\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            printf("%d ", sum[i][j]);
        }
        printf("\n");
    }
}

OUTPUT
Enter 4 numbers for the first matrix:
1 2
3 4
Enter 4 numbers for the second matrix:
5 6
7 8

The Sum is:
6 8 
10 12 

Question 4: Enter 2 matrix from user and subtract them ?

SOLUTION
#include < stdio.h >
int main() {
    int A[2][2], B[2][2], C[2][2];
    int i, j;
    printf("Enter 4 numbers for Matrix A:\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            scanf("%d", &A[i][j]);
        }
    }
    printf("Enter 4 numbers for Matrix B:\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            scanf("%d", &B[i][j]);
        }
    }
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            C[i][j] = A[i][j] - B[i][j];
        }
    }
    printf("\n Subtraction:\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            printf("%d ", C[i][j]);
        }
        printf("\n"); 
     }
}

output

5 8
7 3
Enter 4 numbers for Matrix B:

4 3

Subtraction:

3 0 

Question 5: Enter 2 matrix from user and multiply them ?

SOLUTION
#include < stdio.h >
int main() {
    int A[2][2], B[2][2], C[2][2];
    int i, j, k;
    printf("Enter 4 elements for First Matrix (A):\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            scanf("%d", &A[i][j]);
        }
    }
    printf("Enter 4 elements for Second Matrix (B):\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            scanf("%d", &B[i][j]);
        }
    }
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            C[i][j] = 0; 
            for(k = 0; k < 2; k++) {
                C[i][j] = A[i][k] * B[k][j];
            }
        }
    }

    printf("\nResult Matrix (A * B):\n");
    for(i = 0; i < 2; i++) {
        for(j = 0; j < 2; j++) {
            printf("%d ", C[i][j]);
        }
        printf("\n");
    }
}

OUTPUT
Enter 4 elements for First Matrix (A):
1 2
3 4
Enter 4 elements for Second Matrix (B):
5 6
7 8

Result (A * B):
19 22 
43 50
