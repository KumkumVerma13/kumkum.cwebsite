Question 5: Write a program to enter 2 numbers from user add the number and show the answer , again take 2 number from user subtract the number display the answer ,again take 2 number from user multiply the number display the answer and ,again take 2 number from the user divide the number display the answer ,again take 2 number find the remainder and display the answer ?

SOLUTION:
#include < stdio.h >
int main() {
   int num1, num2;
   int sum, sub, rem, mul;
   float div;

    printf("Enter two number: "); 
    scanf("%d", &num1);
    scanf("%d", &num2);

    sum = num1 + num2;
    printf("\n Addition of two numbers=%d", sum);

    sub = num1 - num2;
    printf("\n Subtraction of two numbers=%d", sub); 

    div = (float)num1 / num2;
    printf("\n Division of two numbers= %f", div);

    rem = num1 % num2;
    printf("\n Remainder of two numbers= %d", rem);

    mul = num1 * num2;
    printf("\n Multiply of two numbers= %d", mul);
}

OUTPUT:
Enter two number: 10 3

 Addition of two numbers=13
 Subtraction of two numbers=7
 Division of two numbers= 3.333333
 Remainder of two numbers= 1
 Multiply of two numbers= 30

Question 8: Enter marks of students and calculate average of student’s marks ?

SOLUTION:
#include < stdio.h >

int main() {
    int n, i;
    float marks, sum = 0, average;

    printf("Enter the number of students: ");
    scanf("%d", &n);

    for (i = 1; i <= n; i++) {
        printf("Enter marks of student %d: ", i);
        scanf("%f", &marks);
        
        sum = sum + marks;
    }

    average = sum / n;

    printf("\nTotal Sum of Marks = %f\n", sum);
    printf("Average Marks = %f\n", average);
}

OUTPUT:
Enter the number of students: 3
Enter marks of student 1: 85
Enter marks of student 2: 90
Enter marks of student 3: 78

Total Sum of Marks = 253.00
Average Marks = 84.33

Quesion 9: Calculate simple interest = (P x R x T)/ 100

SOLUTION:
#include 

int main() {
    float principal, rate, time, simpleinterest;

    printf("Enter Principal: ");
    scanf("%f", &principal);

    printf("Enter Rate : ");
    scanf("%f", &rate);

    printf("Enter Time : ");
    scanf("%f", &time);

    simpleinterest = (principal * rate * time) / 100;
    printf("\nThe Simple Interest is: %f\n", simpleinterest);
}

OUTPUT:
Enter Principal: 1000
Enter Rate : 5
Enter Time : 2

The Simple Interest is: 100.00
