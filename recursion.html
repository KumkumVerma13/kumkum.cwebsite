Question 1: Write a program of recursion and display the output?

Solution:
#include < stdio.h >

int i = 1; 

void display() {
    if (i <= 5) {
        printf("%d ", i); 
        i++;          
        display();  
    }
}

int main() {
    display();
   }
Output:
1 2 3 4 5
 
Question 2:Write a program of recursion with global variable and display the output.

Solution:
#include < stdio.h >
int count = 1; 

void printHello() {
    if (count > 3) {
        return;
    }
    printf("Hello world\n", count);
    count++; 
    printHello();
}

int main() {
    printHello();
}

Output:
Hello world
Hello world
Hello world
