Question 1: Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form ?

Solution:
#include < stdio.h >
#include < conio.h >

struct employee
{
    int id;
    char name[20];
    char designation[20];
};

void main()
{
    struct employee emp[50];
    int i;

    clrscr();

    for(i = 0; i < 50; i++)
    {
        printf("\nEnter ID: ");
        scanf("%d", &emp[i].id);

        printf("Enter Name: ");
        scanf("%s", emp[i].name);

        printf("Enter Designation: ");
        scanf("%s", emp[i].designation);
    }

    printf("\nID\tName\tDesignation");

    for(i = 0; i < 50; i++)
    {
        printf("\n%d\t%s\t%s",
        emp[i].id, emp[i].name, emp[i].designation);
    }
}

Output
Enter ID: 101
Enter Name: Rahul
Enter Designation: Manager

Enter ID: 102
Enter Name: Aman
Enter Designation: Clerk

ID      Name    Designation
101     Rahul   Manager
102     Aman    Clerk
